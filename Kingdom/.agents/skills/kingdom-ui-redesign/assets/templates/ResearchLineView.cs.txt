using UnityEngine;
using UnityEngine.UI;

public sealed class ResearchLineView : MonoBehaviour
{
    [SerializeField] private Image image;
    [SerializeField] private RectTransform rectTransform;

    public void Bind(Vector2 begin, Vector2 end)
    {
        Vector2 delta = end - begin;
        rectTransform.sizeDelta = new Vector2(delta.magnitude, rectTransform.sizeDelta.y);
        rectTransform.localRotation = Quaternion.Euler(0f, 0f, Mathf.Atan2(delta.y, delta.x) * Mathf.Rad2Deg);
        rectTransform.localPosition = begin + delta * 0.5f;
    }

    public void SetColor(Color value)
    {
        image.color = value;
    }
}
